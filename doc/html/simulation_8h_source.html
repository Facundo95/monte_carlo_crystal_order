26struct BoltzmannDeltaETable {
27 // Use quantized integer keys to avoid exact-bit mismatches caused by
28 // floating point rounding differences between the precompute and runtime.
29 std::unordered_map<long long, double> table;
30 double eps; // quantization resolution used to generate keys
31 double T; // temperature used to compute Boltzmann factors (for fallback)
32
33 static inline long long key_of(double x, double eps) {
34 return static_cast<long long>(std::llround(x / eps));
35 }
36
37 BoltzmannDeltaETable(const std::vector<double>& all_dEs, double T_, double eps_ = 1e-6)
38 : eps(eps_), T(T_)
39 {
40 if (T == 0.0) T = 1e-12;
41 for (double dE : all_dEs) {
42 long long key = key_of(dE, eps);
43 table[key] = std::exp(-dE / T);
44 }
45 }
46
47 inline double lookup(double dE) {
48 long long key = key_of(dE, eps);
49 auto it = table.find(key);
50 if (it != table.end()) return it->second;
51
52 // Fallback: compute Boltzmann factor on the fly and insert it
53 double Tloc = T;
54 if (Tloc == 0.0) Tloc = 1e-12;
55 double val = std::exp(-dE / Tloc);
56 table[key] = val;
57 return val;
58 }
59};
